- name: Manage SCCM tasks
  hosts: sccm_console_server
  gather_facts: no
  tasks:
    - name: Add device to SCCM collection
      ansible.windows.win_shell: |
        Import-Module ($Env:SMS_ADMIN_UI_PATH.Substring(0,$Env:SMS_ADMIN_UI_PATH.Length-5) + '\ConfigurationManager.psd1')
        cd "X:\\"  # Replace with your SCCM site code drive
        Add-CMDeviceCollectionDirectMembershipRule -CollectionName "YourCollectionName" -ResourceName "{{ device_name }}"

    - name: Check deployment status
      ansible.windows.win_shell: |
        Import-Module ($Env:SMS_ADMIN_UI_PATH.Substring(0,$Env:SMS_ADMIN_UI_PATH.Length-5) + '\ConfigurationManager.psd1')
        cd "X:\\"  # Replace with your SCCM site code drive
        Get-CMDeploymentStatus -CollectionName "YourCollectionName" | Where-Object { $_.MachineName -eq "{{ device_name }}" }
      register: deployment_status

    - debug:
        var: deployment_status.stdout
